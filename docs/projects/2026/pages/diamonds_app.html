def server(input, output, session):

    @reactive.Calc
    def generate_input_df():
        carat = input.carat()
        cut = input.cut()
        color = input.color()
        clarity = input.clarity()

        large = int((carat > 2))
        log_carat = np.log(carat)
        carat_large = log_carat * large
        if clarity in ['I3', 'I2']:
            clarity = 'I1'

        x_vals = pd.DataFrame({
            'carat': carat,
            'cut': cut,
            'color': color,
            'clarity': clarity,
            'large': large,
            'carat:large': carat_large
        }, index=[0])
        return x_vals

    @reactive.Calc
    def predict_price():
        x_vals = generate_input_df()
        y_pred = mpipe.predict(x_vals)
        pred_val = y_pred[0][0]
        return pred_val
    
    @reactive.Calc
    def generate_ci_vals():
        input_df = generate_input_df()
        percent = input.percent()

        col_transformer = mpipe[:-1]
        x_vals_encoded = col_transformer.transform(input_df)
        
        if type(x_vals_encoded) != np.ndarray:
            x_vals_encoded = x_vals_encoded.toarray()
        xc = sm.add_constant(x_vals_encoded, has_constant='add')
        xc_feature_names = ['intercept'] + list(col_transformer.get_feature_names_out())
        xc_df = pd.DataFrame(xc, columns=xc_feature_names)

        sm_pred = (sm_model.get_prediction(xc_df).summary_frame(alpha=(1-percent)))

        sm_log_mean = sm_pred['mean'].values[0]
        sm_log_ci_low = sm_pred['obs_ci_lower'].values[0]
        sm_log_ci_up = sm_pred['obs_ci_upper'].values[0]
        sm_log_se = (sm_log_ci_up - sm_log_mean) / stats.norm.ppf(1-((1-percent)/2))

        sm_mean = np.exp(sm_log_mean)
        sm_ci_low = np.exp(sm_log_ci_low)
        sm_ci_up = np.exp(sm_log_ci_up)
        plot_vals = [sm_ci_low, sm_mean, sm_ci_up]

        z = np.arange(-4, 4, 0.01)
        pred_vals = np.exp(sm_log_mean + (sm_log_se * z))
        prob = stats.lognorm.pdf(pred_vals, s=sm_log_se, scale=sm_mean)
        plot_probs = stats.lognorm.pdf(plot_vals, s=sm_log_se, scale=sm_mean)

        ci_vals = {
            'pred_vals': pred_vals, 
            'prob': prob,
            'plot_vals': plot_vals,
            'plot_probs': plot_probs
        }
        return ci_vals

    @output
    @render.ui
    def print_price():
        val = predict_price()
        return ui.markdown(f'${val:,.2f}')
    
    @output
    @render.plot
    def ci_plot():
        ci_vals = generate_ci_vals()
        pred_vals = ci_vals['pred_vals']
        prob = ci_vals['prob']
        plot_vals = ci_vals['plot_vals']
        plot_probs = ci_vals['plot_probs']

        plt.figure(figsize=(6,4))
        sns.lineplot(x=pred_vals, y=prob)
        plt.fill_between(pred_vals, prob, where=(pred_vals >= plot_vals[0]) & (pred_vals <= plot_vals[2]), alpha=0.3)
        plt.yticks([])
        plt.xticks(plot_vals, labels=[f'${v:,.0f}' for v in plot_vals])
        plt.tick_params('x', rotation=45)
        plt.vlines([plot_vals], ymin=0, ymax=[plot_probs], color=['gray', 'black', 'gray'], linestyles='--')

    @output
    @render.ui
    def ci_summary():
        percent = input.percent()
        plot_vals = generate_ci_vals()['plot_vals']
        return ui.markdown(f'At the {(percent*100):.0f}% confidence level, the price should fall between: ${plot_vals[0]:,.2f} - ${plot_vals[2]:,.2f}')


app = App(app_ui, server)
